        \item \texttt{long help flag} - Tests --help flag behavior and output
        \item \texttt{short help flag} - Tests -h flag behavior and output

\end{itemize}

    \begin{itemize}

    \end{itemize}
\end{itemize}

\subsubsection*{\texttt{TestInvalidCommand}}
This test validates error handling for invalid CLI commands and graceful failure handling.
\begin{itemize}
    \item \textbf{Tests}:
    \begin{itemize}
        \item \texttt{invalid command} - Tests response to unrecognized commands
        \item \texttt{typo in command} - Tests response to misspelled commands
        \item \texttt{empty args} - Tests response when no arguments provided
    \end{itemize}
\end{itemize}
